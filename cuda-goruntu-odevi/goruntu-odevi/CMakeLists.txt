cmake_minimum_required(VERSION 3.24)
project(GoruntuOdevi LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
find_package(CUDAToolkit REQUIRED)
add_executable(odev main.cpp gpu.cu)
target_include_directories(odev PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(odev PRIVATE ${OpenCV_LIBS} CUDA::cudart)
set_target_properties(odev PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_compile_options(odev PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
if(MSVC)
    target_compile_options(odev PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
